# One executable per suite, grouped by language, so a language that is turned
# off takes its suites with it. `test_core` covers what every build shares.
add_library(script_testmain STATIC test_main.cpp)
target_link_libraries(script_testmain PUBLIC script)
target_include_directories(script_testmain PUBLIC ${PROJECT_SOURCE_DIR}/include ${SCRIPT_LIBCORE_DIR}/libbase/tests ${SCRIPT_LIBCORE_DIR}/libbench/include)
script_set_warnings(script_testmain)

function(script_test_suites group)
    foreach(name ${ARGN})
        set(target test_${group}_${name})
        add_executable(${target} ${group}/test_${name}.cpp)
        target_link_libraries(${target} PRIVATE script_testmain)
        script_set_warnings(${target})
        add_test(NAME ${group}.${name} COMMAND ${target})
        set_tests_properties(${group}.${name} PROPERTIES TIMEOUT 900)
    endforeach()
endfunction()
