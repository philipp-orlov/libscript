# Every example is one file, one executable. Drop a .cpp into a language's
# directory and it is picked up; a language that is turned off takes its
# directory with it. `polyglot/` needs the whole set.
function(script_examples dir)
    file(GLOB sources CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/${dir}/*.cpp)
    foreach(source ${sources})
        get_filename_component(name ${source} NAME_WE)
        set(target example_${dir}_${name})
        add_executable(${target} ${source})
        set_target_properties(${target} PROPERTIES OUTPUT_NAME ${dir}_${name})
        target_link_libraries(${target} PRIVATE script)
        script_set_warnings(${target})
    endforeach()
endfunction()

if(SCRIPT_LANG_CSHARP)
    script_examples(csharp)
endif()
if(SCRIPT_LANG_TYPESCRIPT)
    script_examples(typescript)
endif()
if(SCRIPT_LANG_JAVASCRIPT)
    script_examples(javascript)
endif()
if(SCRIPT_LANG_PYTHON)
    script_examples(python)
endif()
if(SCRIPT_LANG_C)
    script_examples(c)
endif()
if(SCRIPT_LANG_CPP)
    script_examples(cpp)
endif()
if(SCRIPT_LANG_CSHARP AND SCRIPT_LANG_TYPESCRIPT AND SCRIPT_LANG_JAVASCRIPT
   AND SCRIPT_LANG_PYTHON AND SCRIPT_LANG_C AND SCRIPT_LANG_CPP)
    script_examples(polyglot)
endif()
